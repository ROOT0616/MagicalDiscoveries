namespace = magical_discoveries

# One-time beta2 -> beta3 story-state migration.
# Fired before the normal story gatekeeper from on_yearly_pulse_country.
country_event = {
	id = magical_discoveries.90
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = md_beta3_story_migration_done }
	}

	immediate = {
		md_migrate_beta2_story_flags = yes
	}
}

# Yearly gatekeeper.
country_event = {
	id = magical_discoveries.100
	hide_window = yes
	is_triggered_only = yes

	trigger = { md_can_begin_magical_awakening = yes }

	immediate = {
		set_country_flag = md_magic_discovery_offered
		country_event = { id = magical_discoveries.101 }
	}
}

# The first reproducible magical phenomenon.
country_event = {
	id = magical_discoveries.101
	title = magical_discoveries.101.name
	desc = magical_discoveries.101.desc
	picture = GFX_evt_psionics
	is_triggered_only = yes

	option = {
		name = magical_discoveries.101.a
		set_country_flag = md_magic_awakening_started
		start_situation = {
			type = situation_md_magical_awakening
			target = root
		}
		ai_chance = { factor = 9 }
	}
	option = {
		name = magical_discoveries.101.b
		# Rejection is no longer permanent. After ten years, the anomaly can
		# be reconsidered if the empire still exists.
		remove_country_flag = md_magic_discovery_offered
		set_timed_country_flag = {
			flag = md_magic_awakening_suspended
			days = 3600
		}
		ai_chance = { factor = 1 }
	}
}

# 25%: magic is proven to be a reproducible natural phenomenon.
situation_event = {
	id = magical_discoveries.102
	title = magical_discoveries.102.name
	desc = magical_discoveries.102.desc
	picture = GFX_evt_psionics
	is_triggered_only = yes
	situation = this

	immediate = {
		owner = { md_unlock_magic_stage_1 = yes }
	}

	option = {
		name = magical_discoveries.102.a
		custom_tooltip = md_magic_science_granted_tt
		owner = { set_country_flag = md_magic_interpretation_scientific }
	}
	option = {
		name = magical_discoveries.102.b
		custom_tooltip = md_magic_science_granted_tt
		trigger = {
			owner = {
				OR = {
					has_ethic = ethic_spiritualist
					has_ethic = ethic_fanatic_spiritualist
				}
			}
		}
		owner = { set_country_flag = md_magic_interpretation_traditional }
	}
}
