namespace = magical_discoveries

# Yearly gatekeeper. Fired from on_yearly_pulse_country.
country_event = {
	id = magical_discoveries.100
	hide_window = yes
	is_triggered_only = yes

	trigger = { md_can_begin_magical_awakening = yes }

	immediate = {
		set_country_flag = md_magic_discovery_offered
		country_event = { id = magical_discoveries.101 }
	}
}

# The first reproducible magical phenomenon.
country_event = {
	id = magical_discoveries.101
	title = magical_discoveries.101.name
	desc = magical_discoveries.101.desc
	picture = GFX_evt_psionics
	is_triggered_only = yes

	option = {
		name = magical_discoveries.101.a
		set_country_flag = MeetMagicFlag
		start_situation = {
			type = situation_md_magical_awakening
			target = root
		}
		ai_chance = { factor = 9 }
	}
	option = {
		name = magical_discoveries.101.b
		set_country_flag = DenyMagicFlag
		ai_chance = { factor = 1 }
	}
}

country_event = {
	id = magical_discoveries.110

	is_triggered_only = yes

	trigger = { md_can_offer_necronomicon = yes }

	immediate = { country_event = { id = magical_discoveries.111 } }
}

# The Necronomicon. This is the deliberate point of no return for the Heaven scenario.
country_event = {
	id = magical_discoveries.111
	title = magical_discoveries.111.name
	desc = magical_discoveries.111.desc
	picture = GFX_evt_shroud_tunnel
	is_triggered_only = yes

	option = {
		name = magical_discoveries.111.a
		trigger = {
			resource_stockpile_compare = { resource = energy value >= 3500 }
		}
		add_resource = { energy = -3500 }
		md_accept_necronomicon = yes
		ai_chance = { factor = 3 }
	}
	option = {
		name = magical_discoveries.111.b
		set_country_flag = DenyNecronomiconFlag
		ai_chance = { factor = 1 }
	}
}
